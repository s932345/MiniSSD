cmake_minimum_required(VERSION 3.20)
project(MiniSSD VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)

add_library(minissd_storage src/nand_flash.cpp src/ftl.cpp src/workloads.cpp)
target_include_directories(minissd_storage PUBLIC include)
target_compile_features(minissd_storage PUBLIC cxx_std_20)

function(minissd_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
    endif()
endfunction()

minissd_warnings(minissd_storage)

option(MINISSD_SANITIZERS "Enable address/undefined sanitizers on Linux GCC/Clang" OFF)
if(MINISSD_SANITIZERS)
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer option requires Linux GCC or Clang")
    endif()
    target_compile_options(minissd_storage PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(minissd_storage PUBLIC -fsanitize=address,undefined)
endif()

add_executable(minissd src/main.cpp)
target_link_libraries(minissd PRIVATE minissd_storage)
if(MINGW)
    target_link_options(minissd PRIVATE -municode)
endif()
minissd_warnings(minissd)

include(CTest)
if(BUILD_TESTING)
    add_executable(nand_tests tests/test_nand.cpp)
    target_link_libraries(nand_tests PRIVATE minissd_storage)
    minissd_warnings(nand_tests)
    add_test(NAME virtual_nand COMMAND nand_tests)
    add_executable(ftl_tests tests/test_ftl.cpp)
    target_link_libraries(ftl_tests PRIVATE minissd_storage)
    minissd_warnings(ftl_tests)
    add_test(NAME page_ftl COMMAND ftl_tests)
    add_executable(gc_tests tests/test_gc.cpp)
    target_link_libraries(gc_tests PRIVATE minissd_storage)
    minissd_warnings(gc_tests)
    add_test(NAME garbage_collection COMMAND gc_tests)
    add_test(NAME validation_cli COMMAND minissd validate --seed 42 --operations 5000)
    add_test(NAME benchmark_cli COMMAND minissd benchmark --seed 42 --writes 1000)
    add_test(NAME unicode_report_cli COMMAND minissd validate --seed 0 --operations 1
        --report "${CMAKE_CURRENT_BINARY_DIR}/report-驗證.json")
endif()
